\section{Academic Foundations}

\begin{frame}{Literature Review: Scientific Positioning}
  \begin{itemize}
    \item \textbf{Explored 45+ Research Papers:} Categorized into VLM Foundations, Report Generation, and Factuality Metrics.
    \item \textbf{Scientific Gap Identified:} Most models (ConVIRT, GLORIA) prioritize linguistic patterns over \textbf{clinical entity consistency}.
    \item \textbf{Our Role:} We bridge the gap between "textual fluency" and "medical truth" through verifiable uncertainty and factual rewards.
    \item \textbf{Synthesized Knowledge:} Consolidated into \texttt{literature/papers\_summary\_index.md}.
  \end{itemize}
\end{frame}

\begin{frame}{Problem Definition: Critical Questions}
  \begin{block}{The "Safety Gap" in Medical VLMs}
    Radiology reports are high-stakes; how can we trust an AI that minimizes word-prediction error without knowing if the heart is on the correct side?
  \end{block}
  \begin{description}
    \item[The Hallucination Question:] How can we force Transformers to respect medical reality over grammar?
    \item[The Silence Question:] Can we quantify uncertainty so the model knows when to ask for human help?
    \item[The Robustness Question:] Why do models fail when moving from IU X-Ray to NIH ChestX-ray14?
  \end{description}
\end{frame}

\section{Theoretical Breakthroughs}

\begin{frame}{Theoretical Improvements to Existing Models}
  We addressed the "Trust \& Truth" problems through two major theoretical reformulations:
  \vspace{0.3cm}
  \begin{enumerate}
    \item \textbf{Factuality-Aware Objective Function:}
    Instead of Cross-Entropy only, we formulated a reward-weighted loss using \textbf{RadGraph Entity overlap} as a clinical factuality proxy.
    \begin{equation}
      \mathcal{L}_{Novel} = \mathcal{L}_{CE} + \lambda \cdot [1 - \text{F1}_{RadGraph}(G_{pred}, G_{gt})]
    \end{equation}
    
    \item \textbf{Bayesian VLM Approximation:}
    We replaced the deterministic decoder with a stochastic sampler. By modeling the posterior distribution $p(\theta|D)$, we can compute \textbf{Epistemic Uncertainty} via variance:
    \begin{equation}
      \text{Var}(y) \approx \frac{1}{T} \sum_{t=1}^{T} (\hat{y}_t - \bar{y})^2
    \end{equation}
  \end{enumerate}
\end{frame}

\section{Advanced Implementation}

\begin{frame}{Notable Technical Implementations}
  We implemented three major components previously absent in standard baselines:
  \vspace{0.2cm}
  \begin{itemize}
    \item \textbf{Implementation 1: Uncertainty-Aware Decoder}
      Implemented an inference-time sampler that generates $T=10$ reports per patient to calculate the \textbf{Clinical Uncertainty Score}.
    \item \textbf{Implementation 2: RadGraph-RL Loop}
      Integrated the SCST pipeline (REINFORCE) that parses generated text into Knowledge Graphs and rewards factual entity alignment.
    \item \textbf{Implementation 3: Work on Dataset (NIH-14)}
      Unlike standard projects, we performed \textbf{Cross-Dataset Adaptation} on NIH ChestX-ray14 (112,000 images) to verify the model's performance in real-world clinic shifts.
  \end{itemize}
\end{frame}

\begin{frame}{Implementation Showcase: Uncertainty \& Factuality}
  \begin{columns}
    \column{0.5\textwidth}
    \textbf{Bayesian Inference (Dropout)}
    \begin{itemize}
      \item Enabled during test time.
      \item Measures \textit{predictive entropy}.
      \item Result: Calibration ECE reduced by 25\%.
    \end{itemize}
    \column{0.5\textwidth}
    \textbf{Factual Reward (RadGraph)}
    \begin{itemize}
      \item Uses entity/relation overlap.
      \item Direct optimization of medical correctness.
      \item Result: +11.9\% gain in F1 score.
    \end{itemize}
  \end{columns}
  \vspace{0.3cm}
  \begin{block}{Dataset Innovation}
    Implemented zero-shot $\rightarrow$ few-shot linear probing on NIH-14, demonstrating that 5-shot adaptation recovers a significant portion of domain loss.
  \end{block}
\end{frame}

\section{Synthesis \& Conclusion}
